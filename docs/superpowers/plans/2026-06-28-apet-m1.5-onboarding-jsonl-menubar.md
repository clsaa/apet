# apet M1.5 实现计划：开箱即用 + 常规 App 体验

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 apet 不装 hook 也能看到所有 Claude Code 会话（含当前在跑的），并提供常规 macOS App 的菜单/引导体验。

**Architecture:** 新增 jsonl 只读扫描源，用**内容信号**（stop_reason/away_summary/queue-operation/尾元数据）派生状态，经**已有的 `NDJSONIngestor`（唯一 seq 源）**合成事件融进**已有的 `SessionStore`**；菜单栏补右键 NSMenu + 左键面板齿轮；引导精简 + hook/通知授权 just-in-time。

**Tech Stack:** Swift 5.9 / SwiftPM 多 target（AgentPetCore 纯逻辑、AppShellKit 可测胶水、apet GUI）、XCTest、Foundation only。

## Global Constraints

权威设计：`docs/superpowers/specs/2026-06-28-apet-onboarding-jsonl-menubar-design.md`。以下 verbatim 自 CLAUDE.md 与该 spec：

- **零外部依赖**——只用 Foundation 标准库。
- **不用 `Date()` / `Date.now`**——"当前时间"一律通过参数 `now: Double`（Unix 秒，墙钟，与 fs mtime 同域）注入。
- **排序唯一事实是 `seq`**，由**唯一的 `NDJSONIngestor` 实例**按 append 顺序赋；jsonl 路径**必须复用同一 ingestor 取号**，不得自带计数器（架构-B2）。去重唯一键 `eventId`；归一键 `(agent, root, sessionId)` 必须含 `root`。
- **终态 `ended` 不可回退**；`stale` 可复活。
- **字段级合并**（last-non-nil-wins）：`cwd/title/terminal` 空值不覆盖已有值；`terminal` 精确值不被空值降级。
- **测试是规范**——测试失败改实现、不改测试。断言精确，覆盖正常/边界/异常。
- **不动 wire 协议**：jsonl 是进程内合成事件，不写 events.ndjson、不进 `AgentEvent.decode`、不加 `source` 字段。消费端用 `Session.terminal == nil` 判"能否精确跳转"。
- **窗口默认**：`runningWindow = 120`、`idleWindow = 1800`（30min）。运算符方向：running `now-effectiveTs < runningWindow`；waitingStop `[runningWindow, idleWindow)`；ignore/stale `≥ idleWindow`。窗口作入参注入。
- **分支/推送**：feature 分支开发；push 用 SSH `git@github.com:clsaa/apet.git`；身份 `clsaa <812022339@qq.com>`；中文 commit。

private func showRightClickMenu() {
    guard let button = statusItem.button else { return }
    let summary = summaryProvider?() ?? (running: 0, waiting: 0)
    let menu = NSMenu()
    for row in MenuBarMenuModel.rows(runningCount: summary.running, waitingCount: summary.waiting) {
        if row.command == .sessionSummary {
            let item = NSMenuItem(title: row.title, action: nil, keyEquivalent: "")
            item.isEnabled = false
            menu.addItem(item)
            menu.addItem(.separator())
            continue
        }
        let item = NSMenuItem(title: row.title,
                              action: #selector(handleMenuCommand(_:)),
                              keyEquivalent: row.shortcut ?? "")
        item.target = self
        item.representedObject = row.command
        menu.addItem(item)
    }
    button.highlight(true)
    menu.popUp(positioning: nil, at: NSPoint(x: 0, y: button.bounds.height + 4), in: button)
    button.highlight(false)

@objc private func handleMenuCommand(_ sender: NSMenuItem) {
    guard let cmd = sender.representedObject as? MenuCommand else { return }

    case .preferences: onOpenPreferences?()
    case .quit: NSApplication.shared.terminate(nil)

```swift
import Foundation

public enum TailReadResult: Equatable {
    case ok(lines: [String])
    case unreadable(path: String)
}

public enum TailLineReader {
    public static func lastLines(path: String, maxLines: Int, maxBytes: Int) -> TailReadResult {
        guard let handle = FileHandle(forReadingAtPath: path) else { return .unreadable(path: path) }
        defer { try? handle.close() }
        do {
            let size = try handle.seekToEnd()
            let readLen = min(UInt64(max(maxBytes, 0)), size)
            let start = size - readLen
            try handle.seek(toOffset: start)
            let data = try handle.read(upToCount: Int(readLen)) ?? Data()
            var text = String(decoding: data, as: UTF8.self)
            if start > 0, let nl = text.firstIndex(of: "\n") {
                text = String(text[text.index(after: nl)...])   // 丢首个残行
            }
            let lines = text.split(separator: "\n", omittingEmptySubsequences: true).map(String.init)
            return .ok(lines: Array(lines.suffix(max(maxLines, 0))))
        } catch {
            return .unreadable(path: path)
        }
    }

    public static func firstLine(path: String) -> String? {
        guard let handle = FileHandle(forReadingAtPath: path) else { return nil }
        defer { try? handle.close() }
        // 读前 64KB 足够覆盖首条记录
        let data = (try? handle.read(upToCount: 65_536)) ?? nil
        guard let data else { return nil }
        let text = String(decoding: data, as: UTF8.self)
        return text.split(separator: "\n", omittingEmptySubsequences: true).first.map(String.init)
    }
}
```

import Foundation

public struct ScannedFile: Equatable {
    public var sessionId: String, root: String
    public var cwd: String?, title: String?, lastPrompt: String?
    public var mtime: Double
    public var lastAssistantStopReason: String?
    public var hasAwaySummary: Bool, hasRecentQueueOp: Bool
    public var lastConversationTs: Double?
    public var entrypoint: String?, promptSource: String?
    public var isSidechain: Bool, isSubagentPath: Bool
    public init(sessionId: String, root: String, cwd: String? = nil, title: String? = nil,
                lastPrompt: String? = nil, mtime: Double, lastAssistantStopReason: String? = nil,
                hasAwaySummary: Bool = false, hasRecentQueueOp: Bool = false,
                lastConversationTs: Double? = nil, entrypoint: String? = nil,
                promptSource: String? = nil, isSidechain: Bool = false, isSubagentPath: Bool = false) {
        self.sessionId = sessionId; self.root = root; self.cwd = cwd; self.title = title
        self.lastPrompt = lastPrompt; self.mtime = mtime
        self.lastAssistantStopReason = lastAssistantStopReason
        self.hasAwaySummary = hasAwaySummary; self.hasRecentQueueOp = hasRecentQueueOp
        self.lastConversationTs = lastConversationTs; self.entrypoint = entrypoint
        self.promptSource = promptSource; self.isSidechain = isSidechain; self.isSubagentPath = isSubagentPath
    }
}
public enum SyntheticKind: Equatable { case sdkCli, sdkPromptSource }
public enum IgnoreReason: Equatable { case subagent, synthetic(SyntheticKind), blacklistedCwd, tooOld }
public enum ScanState: Equatable { case running, waitingStop, stale }
public enum ScanResult: Equatable {
    case observe(state: ScanState, key: SessionKey, cwd: String?, title: String?)
    case ignore(IgnoreReason)
}

public enum JSONLSessionScanner {
    static let cwdBlacklist = ["-iteration-", "/eval-", "/private/tmp/claude-", "/private/var/folders"]

    public static func scan(_ f: ScannedFile, now: Double, priorState: ScanState?,
                            runningWindow: Double = 120, idleWindow: Double = 1800) -> ScanResult {
        // 过滤优先级：subagent > synthetic > 黑名单 > tooOld
        if f.isSubagentPath || f.isSidechain { return .ignore(.subagent) }
        if f.entrypoint == "sdk-cli" { return .ignore(.synthetic(.sdkCli)) }
        if f.promptSource == "sdk" { return .ignore(.synthetic(.sdkPromptSource)) }
        if let cwd = f.cwd, cwdBlacklist.contains(where: { cwd.contains($0) }) {
            return .ignore(.blacklistedCwd)
        }
        let effectiveTs = min(f.mtime, f.lastConversationTs ?? f.mtime)
        let age = now - effectiveTs
        if age >= idleWindow { return .ignore(.tooOld) }
        // 状态派生在 Task 5 补全；此处占位
        let key = SessionKey(agent: "claude", root: f.root, sessionId: f.sessionId)
        let title = f.title ?? f.lastPrompt
        return .observe(state: .stale, key: key, cwd: f.cwd, title: title)
    }
}

public func markStaleSession(_ key: SessionKey, now: Double) -> [StoreChange] {
    let changes = markStaleSessionInner(key, now: now)
    if !changes.isEmpty { fanOut(changes, replay: false) }   // 与既有扇出一致

private func markStaleSessionInner(_ key: SessionKey, now: Double) -> [StoreChange] {
    guard var s = sessions[key] else { return [] }

    return [.upserted(key)]

public enum NotificationStatus: Equatable { case authorized, denied, notDetermined }
public enum HookStatus: Equatable { case installed(path: String), notInstalled, failed(reason: String) }
public enum JSONLSourceStatus: Equatable { case found(count: Int), pathMissing, unreadable(path: String) }
public enum OverallHealth: Equatable { case ready, readyEnhanced, degraded(reason: String) }

public struct ConfigHealth: Equatable {
    public let notification: NotificationStatus
    public let hook: HookStatus
    public let jsonlSource: JSONLSourceStatus
    public let dataRoots: [String]
    public init(notification: NotificationStatus, hook: HookStatus,
                jsonlSource: JSONLSourceStatus, dataRoots: [String]) {
        self.notification = notification; self.hook = hook
        self.jsonlSource = jsonlSource; self.dataRoots = dataRoots

    public var overall: OverallHealth {
        switch jsonlSource {
        case .pathMissing, .unreadable: return .degraded(reason: "jsonl 数据源不可用")
        case .found:

            return .ready
        }
    }
